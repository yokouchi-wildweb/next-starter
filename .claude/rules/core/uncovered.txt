pnpm-workspace.yaml  # ワークスペースの宣言
.gitignore  # git の除外
servers/*/.gitignore  # git の除外(servers の各プログラム)
servers/*/.npmrc  # npm の設定(servers の各プログラム)
.env.example  # 環境変数の見本。値の決まりは appEnv と各 lib の README
.prettierrc  # 整形の設定
.mcp.json  # MCP の接続の設定
components.json  # shadcn の生成の設定
next.config.ts  # Next.js のビルドの設定
postcss.config.mjs  # PostCSS の設定
tsconfig.json  # 型の検査の設定
tsconfig.typecheck.json  # 型の検査の設定
eslint.config.mjs  # lint の設定(規則の中身は eslint-rules/ で scripts の決まりが当たる)
eslint.config.fork.mjs  # fork の lint の設定
drizzle.config.ts  # drizzle-kit の接続の設定
firebase.json  # Firebase の配置の設定
firestore.rules.example  # Firestore の規則の見本
storage.rules  # Firebase Storage の規則(拒否だけ)
patches/**  # 依存への当て布(pnpm patch の出力)
.github/**  # CI の見本
.agents/**  # Codex 向けの skills への案内(.claude/skills へのリンク)
.codex/**  # Codex の設定と案内
